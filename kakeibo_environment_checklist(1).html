docker/app/Dockerfile の内容
FROM node:22-bookworm-slim AS node
FROM composer:2 AS composer
FROM php:8.4-fpm-bookworm

ENV TZ=Asia/Tokyo
ENV COMPOSER_ALLOW_SUPERUSER=1

RUN apt-get update \
    && apt-get install -y --no-install-recommends \
        git unzip curl libicu-dev libzip-dev libonig-dev libxml2-dev default-mysql-client \
    && docker-php-ext-install bcmath intl mbstring pdo_mysql zip opcache \
    && apt-get clean \
    && rm -rf /var/lib/apt/lists/*

COPY --from=composer /usr/bin/composer /usr/local/bin/composer
COPY --from=node /usr/local/bin/node /usr/local/bin/node
COPY --from=node /usr/local/lib/node_modules /usr/local/lib/node_modules
RUN ln -s /usr/local/lib/node_modules/npm/bin/npm-cli.js /usr/local/bin/npm \
    && ln -s /usr/local/lib/node_modules/npm/bin/npx-cli.js /usr/local/bin/npx
COPY docker/app/php.ini /usr/local/etc/php/conf.d/99-household.ini
WORKDIR /var/www/html
CMD ["php-fpm"]

docker/nginx/kakeibo.conf の内容
server {
    listen 80;
    server_name localhost;
    root /var/www/html/public;
    index index.php index.html;
    charset utf-8;
    client_max_body_size 20M;

    location / {
        try_files $uri $uri/ /index.php?$query_string;
    }

    location ~ \.php$ {
        include fastcgi_params;
        fastcgi_pass kakeibo-app:9000;
        fastcgi_index index.php;
        fastcgi_param SCRIPT_FILENAME $document_root$fastcgi_script_name;
    }

    location ~ /\.(?!well-known).* { deny all; }
}

compose.yaml の内容
services:
  kakeibo-app:
    build:
      context: .
      dockerfile: docker/app/Dockerfile
    working_dir: /var/www/html
    volumes:
      - ./src/kakeibo:/var/www/html
      - kakeibo-vendor:/var/www/html/vendor
      - kakeibo-node-modules:/var/www/html/node_modules
    ports:
      - "5173:5173"
    environment:
      TZ: Asia/Tokyo
    depends_on:
      db:
        condition: service_healthy
    networks:
      - household-network

  kakeibo-web:
    image: nginx:stable-alpine
    ports:
      - "8080:80"
    volumes:
      - ./src/kakeibo:/var/www/html:ro
      - ./docker/nginx/kakeibo.conf:/etc/nginx/conf.d/default.conf:ro
    depends_on:
      - kakeibo-app
    networks:
      - household-network

  db:
    image: mysql:8.4
    ports:
      - "3306:3306"
    environment:
      TZ: Asia/Tokyo
      MYSQL_ROOT_PASSWORD: ${DB_ROOT_PASSWORD}
    volumes:
      - mysql-data:/var/lib/mysql
      - ./docker/mysql/init:/docker-entrypoint-initdb.d:ro
    healthcheck:
      test: ["CMD", "mysqladmin", "ping", "-h", "localhost", "-u", "root", "-p${DB_ROOT_PASSWORD}"]
      interval: 5s
      timeout: 5s
      retries: 20
      start_period: 10s
    networks:
      - household-network

networks:
  household-network:
    driver: bridge

volumes:
  mysql-data:
  kakeibo-vendor:
  kakeibo-node-modules:
